/**
 * Generate ElevenLabs Music background track from `src/music.ts`.
 * Hash-cached: regenerates only if the prompt or duration changed.
 *
 *   npm run generate-music
 */

import * as fs from "node:fs";
import * as path from "node:path";
import * as crypto from "node:crypto";
import * as url from "node:url";
import { execSync } from "node:child_process";
import "dotenv/config";
// @ts-expect-error — no types for mp3-duration
import mp3Duration from "mp3-duration";
import { MUSIC, MUSIC_AUDIO_SUBDIR, MUSIC_FILENAME } from "./src/music";

const __dirname = path.dirname(url.fileURLToPath(import.meta.url));
const PUBLIC_DIR = path.resolve(__dirname, "..", "..", "brand", "motion");
const MUSIC_DIR = path.join(PUBLIC_DIR, MUSIC_AUDIO_SUBDIR);
const TIMINGS_PATH = path.join(__dirname, "music.timings.json");

interface Timing {
  hash: string;
  durationSeconds: number;
  generatedAt: string;
}

async function main() {
  const apiKey = process.env.ELEVENLABS_API_KEY;
  if (!apiKey) {
    console.error("Missing ELEVENLABS_API_KEY in environment.");
    process.exit(1);
  }
  fs.mkdirSync(MUSIC_DIR, { recursive: true });

  const file = path.join(MUSIC_DIR, MUSIC_FILENAME);
  const hash = sha(`${MUSIC.prompt}:${MUSIC.durationMs}`);
  const existing: Timing | null = fs.existsSync(TIMINGS_PATH)
    ? (JSON.parse(fs.readFileSync(TIMINGS_PATH, "utf8")) as Timing)
    : null;

  if (existing?.hash === hash && fs.existsSync(file)) {
    console.log(`✓ background.mp3 cached (${existing.durationSeconds.toFixed(2)}s)`);
    return;
  }

  process.stdout.write(`→ generating ${MUSIC.durationMs / 1000}s of music... `);
  const buf = await synthesize(apiKey, MUSIC.prompt, MUSIC.durationMs);
  const rawFile = file.replace(/\.mp3$/, ".raw.mp3");
  fs.writeFileSync(rawFile, buf);

  // Trim leading silence so playback hits the ear right at frame 0.
  // ElevenLabs music often starts with a brief quiet beat that reads as
  // "no music" against narration.
  process.stdout.write("trimming silence... ");
  execSync(
    `ffmpeg -y -i "${rawFile}" -af "silenceremove=start_periods=1:start_duration=0.2:start_threshold=-40dB" -c:a libmp3lame -b:a 128k "${file}" -loglevel error`,
  );
  fs.unlinkSync(rawFile);

  const duration = await measureDuration(file);
  const timing: Timing = { hash, durationSeconds: duration, generatedAt: new Date().toISOString() };
  fs.writeFileSync(TIMINGS_PATH, JSON.stringify(timing, null, 2) + "\n");
  console.log(`done (${duration.toFixed(2)}s)`);
}

function sha(text: string): string {
  return crypto.createHash("sha256").update(text).digest("hex").slice(0, 16);
}

async function synthesize(apiKey: string, prompt: string, durationMs: number): Promise<Buffer> {
  const res = await fetch("https://api.elevenlabs.io/v1/music", {
    method: "POST",
    headers: {
      "xi-api-key": apiKey,
      "Content-Type": "application/json",
      Accept: "audio/mpeg",
    },
    body: JSON.stringify({
      prompt,
      music_length_ms: durationMs,
    }),
  });
  if (!res.ok) {
    const body = await res.text();
    throw new Error(`ElevenLabs Music ${res.status}: ${body.slice(0, 240)}`);
  }
  return Buffer.from(await res.arrayBuffer());
}

function measureDuration(filepath: string): Promise<number> {
  return new Promise((resolve, reject) => {
    mp3Duration(filepath, (err: Error | null, d: number) => (err ? reject(err) : resolve(d)));
  });
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
