/**
 * Generate ElevenLabs sound effects for every clip in `src/sfx.ts`.
 * Hash-cached so unchanged clips skip the API. Mirrors generate-audio.ts.
 *
 *   npm run generate-sfx
 */

import * as fs from "node:fs";
import * as path from "node:path";
import * as crypto from "node:crypto";
import * as url from "node:url";
import "dotenv/config";
// @ts-expect-error — no types for mp3-duration
import mp3Duration from "mp3-duration";
import { SFX, SFX_AUDIO_SUBDIR } from "./src/sfx";

const __dirname = path.dirname(url.fileURLToPath(import.meta.url));
const PUBLIC_DIR = path.resolve(__dirname, "..", "..", "brand", "motion");
const SFX_DIR = path.join(PUBLIC_DIR, SFX_AUDIO_SUBDIR);
const TIMINGS_PATH = path.join(__dirname, "sfx.timings.json");

interface Timing {
  hash: string;
  durationSeconds: number;
  generatedAt: string;
}

async function main() {
  const apiKey = process.env.ELEVENLABS_API_KEY;
  if (!apiKey) {
    console.error("Missing ELEVENLABS_API_KEY in environment.");
    process.exit(1);
  }
  fs.mkdirSync(SFX_DIR, { recursive: true });

  const existing: Record<string, Timing> = fs.existsSync(TIMINGS_PATH)
    ? JSON.parse(fs.readFileSync(TIMINGS_PATH, "utf8"))
    : {};
  const updated: Record<string, Timing> = { ...existing };

  let generated = 0;
  let cached = 0;

  for (const clip of SFX) {
    const hash = sha(`${clip.prompt}:${clip.durationSeconds}`);
    const file = path.join(SFX_DIR, `${clip.id}.mp3`);
    const prev = existing[clip.id];

    if (prev?.hash === hash && fs.existsSync(file)) {
      console.log(`✓ ${clip.id.padEnd(14)} cached  (${prev.durationSeconds.toFixed(2)}s)`);
      cached++;
      continue;
    }

    process.stdout.write(`→ ${clip.id.padEnd(14)} generating... `);
    const buf = await synthesize(apiKey, clip.prompt, clip.durationSeconds);
    fs.writeFileSync(file, buf);
    const duration = await measureDuration(file);
    updated[clip.id] = { hash, durationSeconds: duration, generatedAt: new Date().toISOString() };
    console.log(`done (${duration.toFixed(2)}s)`);
    generated++;
  }

  // Drop entries no longer in SFX list
  const liveIds = new Set(SFX.map((c) => c.id));
  for (const id of Object.keys(updated)) {
    if (!liveIds.has(id)) delete updated[id];
  }

  fs.writeFileSync(TIMINGS_PATH, JSON.stringify(updated, null, 2) + "\n");
  console.log(`\nGenerated: ${generated}  Cached: ${cached}`);
  console.log(`Wrote ${path.relative(process.cwd(), TIMINGS_PATH)}`);
}

function sha(text: string): string {
  return crypto.createHash("sha256").update(text).digest("hex").slice(0, 16);
}

async function synthesize(apiKey: string, prompt: string, durationSeconds: number): Promise<Buffer> {
  const res = await fetch("https://api.elevenlabs.io/v1/sound-generation", {
    method: "POST",
    headers: {
      "xi-api-key": apiKey,
      "Content-Type": "application/json",
      Accept: "audio/mpeg",
    },
    body: JSON.stringify({
      text: prompt,
      duration_seconds: durationSeconds,
      prompt_influence: 0.3,
    }),
  });
  if (!res.ok) {
    const body = await res.text();
    throw new Error(`ElevenLabs ${res.status}: ${body.slice(0, 240)}`);
  }
  return Buffer.from(await res.arrayBuffer());
}

function measureDuration(filepath: string): Promise<number> {
  return new Promise((resolve, reject) => {
    mp3Duration(filepath, (err: Error | null, d: number) => (err ? reject(err) : resolve(d)));
  });
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
