/**
 * Sound effects for the launch video.
 * Each clip is anchored to a specific frame in the visual timeline.
 *
 * Edit a prompt → run `npm run generate-sfx` → only the changed clip
 * regenerates (others cached by hash).
 */

import { beatStartFrame } from "./timings";

export interface SfxClip {
  id: string;
  /** ElevenLabs sound-generation prompt. Be specific; modern/clean/short reads better. */
  prompt: string;
  /** Requested duration in seconds. ElevenLabs returns slightly variable length. */
  durationSeconds: number;
  /** Frame in the composition where this clip starts. */
  fromFrame: number;
  /** 0–1 mix level. Default 0.6. */
  volume?: number;
}
