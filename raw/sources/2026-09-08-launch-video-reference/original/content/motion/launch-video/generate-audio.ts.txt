/**
 * Generate ElevenLabs TTS audio for every beat in `src/script.ts`.
 *
 * Beats are content-addressed by SHA-256 of their text — if the text
 * hasn't changed since the last run the API call is skipped. This makes
 * iteration cheap: edit one beat, regenerate, the others stay cached.
 *
 * Outputs:
 *   - brand/motion/audio/launch-narration/{beat.id}.mp3
 *   - script.timings.json   (hash + measured durations)
 *
 * Run:   npm run generate-audio
 * Env:   ELEVENLABS_API_KEY (required), VOICE_ID, MODEL_ID (optional)
 */

import * as fs from "node:fs";
import * as path from "node:path";
import * as crypto from "node:crypto";
import * as url from "node:url";
import { spawnSync } from "node:child_process";
import "dotenv/config";
// eslint-disable-next-line @typescript-eslint/ban-ts-comment
// @ts-expect-error — mp3-duration ships no .d.ts; inline type is fine for one call.
import mp3Duration from "mp3-duration";
import { SCRIPT, AUDIO_SUBDIR } from "./src/script";

const __dirname = path.dirname(url.fileURLToPath(import.meta.url));

const FPS = 30;

// Default voice picked from the ElevenLabs library. Runs on
// `eleven_v3` so breath/pause tags produce real micro-pauses.
// Override via .env or per-run env vars.
const VOICE_ID = process.env.VOICE_ID || "dXtC3XhB9GtPusIpNtQx";
const MODEL_ID = process.env.MODEL_ID || "eleven_v3";
const OUTPUT_FORMAT = "mp3_44100_128";

// "Natural" settings tuned for v3 + audio tags per ElevenLabs docs:
// lower stability = more variation, higher style = more inflection.
const VOICE_SETTINGS = {
  stability: 0.3,
  similarity_boost: 0.75,
  style: 0.5,
  use_speaker_boost: true,
} as const;

// Speed multiplier applied via ffmpeg `atempo` after synthesis. v3 doesn't
// expose a reliable speed param, and atempo preserves pitch. 1.0 = native.
const SPEED = parseFloat(process.env.SPEED || "1.10");

// On v3, a leading `[breathes]` tag produces a real breath in
// before each beat — single biggest "less AI" lever per docs.
function shapeText(text: string): string {
  if (MODEL_ID.startsWith("eleven_v3")) return `[breathes] ${text}`;
  return text;
}

const PUBLIC_DIR = path.resolve(__dirname, "..", "..", "brand", "motion");
const AUDIO_DIR = path.join(PUBLIC_DIR, AUDIO_SUBDIR);
const TIMINGS_PATH = path.join(__dirname, "script.timings.json");

interface Timing {
  hash: string;
  durationSeconds: number;
  durationFrames: number;
  generatedAt: string;
}
type TimingsMap = Record<string, Timing>;

async function main() {
  const apiKey = process.env.ELEVENLABS_API_KEY;
  if (!apiKey) {
    console.error("Missing ELEVENLABS_API_KEY in environment.");
    console.error("Copy .env.example to .env and fill it in.");
    process.exit(1);
  }

  fs.mkdirSync(AUDIO_DIR, { recursive: true });

  const existing: TimingsMap = fs.existsSync(TIMINGS_PATH)
    ? JSON.parse(fs.readFileSync(TIMINGS_PATH, "utf8"))
    : {};
  const updated: TimingsMap = { ...existing };

  let generated = 0;
  let cached = 0;

  for (const beat of SCRIPT) {
    // Hash includes voice + model + settings + speed so swapping any
    // of them invalidates the cache and forces a re-render.
    const hash = sha(`${beat.text}\x00${VOICE_ID}\x00${MODEL_ID}\x00${JSON.stringify(VOICE_SETTINGS)}\x00${SPEED}`);
    const audioFile = path.join(AUDIO_DIR, `${beat.id}.mp3`);
    const prev = existing[beat.id];

    if (prev && prev.hash === hash && fs.existsSync(audioFile)) {
      console.log(`✓ ${beat.id.padEnd(12)} cached  (${prev.durationSeconds.toFixed(2)}s)`);
      cached++;
      continue;
    }

    process.stdout.write(`→ ${beat.id.padEnd(12)} generating... `);
    const buffer = await synthesize(apiKey, shapeText(beat.text));
    fs.writeFileSync(audioFile, buffer);
    applySpeed(audioFile, SPEED);

    const durationSeconds = await measureDurationSeconds(audioFile);
    const durationFrames = Math.ceil(durationSeconds * FPS);

    updated[beat.id] = {
      hash,
      durationSeconds,
      durationFrames,
      generatedAt: new Date().toISOString(),
    };
    console.log(`done  (${durationSeconds.toFixed(2)}s = ${durationFrames}f)`);
    generated++;
  }

  // Drop entries for beats that no longer exist in SCRIPT.
  const liveIds = new Set(SCRIPT.map((b) => b.id));
  for (const id of Object.keys(updated)) {
    if (!liveIds.has(id)) {
      delete updated[id];
      console.log(`× ${id.padEnd(12)} removed (no longer in SCRIPT)`);
    }
  }

  fs.writeFileSync(TIMINGS_PATH, JSON.stringify(updated, null, 2) + "\n");

  const total = Object.values(updated).reduce(
    (sum, t) => sum + t.durationSeconds,
    0,
  );
  const totalFrames = Object.values(updated).reduce(
    (sum, t) => sum + t.durationFrames,
    0,
  );
  console.log("");
  console.log(`Generated: ${generated}  Cached: ${cached}`);
  console.log(`Total narration: ${total.toFixed(2)}s = ${totalFrames}f`);
  console.log(`Wrote ${path.relative(process.cwd(), TIMINGS_PATH)}`);
}

function sha(text: string): string {
  return crypto.createHash("sha256").update(text).digest("hex").slice(0, 16);
}

async function synthesize(apiKey: string, text: string): Promise<Buffer> {
  const res = await fetch(
    `https://api.elevenlabs.io/v1/text-to-speech/${VOICE_ID}?output_format=${OUTPUT_FORMAT}`,
    {
      method: "POST",
      headers: {
        "xi-api-key": apiKey,
        "Content-Type": "application/json",
        Accept: "audio/mpeg",
      },
      body: JSON.stringify({
        text,
        model_id: MODEL_ID,
        voice_settings: VOICE_SETTINGS,
      }),
    },
  );
  if (!res.ok) {
    const detail = await res.text();
    throw new Error(`ElevenLabs ${res.status}: ${detail.slice(0, 200)}`);
  }
  const arrayBuffer = await res.arrayBuffer();
  return Buffer.from(arrayBuffer);
}

function applySpeed(filepath: string, speed: number): void {
  if (Math.abs(speed - 1) < 0.001) return;
  const tmp = filepath + ".tmp.mp3";
  const res = spawnSync(
    "ffmpeg",
    ["-y", "-loglevel", "error", "-i", filepath, "-af", `atempo=${speed}`, "-c:a", "libmp3lame", "-b:a", "128k", tmp],
    { stdio: "inherit" },
  );
  if (res.status !== 0) throw new Error(`ffmpeg atempo failed (${res.status})`);
  fs.renameSync(tmp, filepath);
}

function measureDurationSeconds(filepath: string): Promise<number> {
  return new Promise((resolve, reject) => {
    mp3Duration(filepath, (err: Error | null, seconds: number) => {
      if (err) reject(err);
      else resolve(seconds);
    });
  });
}

main().catch((err) => {
  console.error(err);
  process.exit(1);
});
