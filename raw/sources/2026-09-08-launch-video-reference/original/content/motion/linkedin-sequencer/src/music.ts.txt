/**
 * Background music track. One continuous clip that plays under everything
 * at a low volume so it doesn't compete with narration.
 *
 * Edit the prompt → run `npm run generate-music` → re-generates if hashed
 * prompt changed.
 */

export interface MusicTrack {
  /** ElevenLabs music prompt. */
  prompt: string;
  /** Track length in milliseconds. ElevenLabs Music API caps at 5 minutes. */
  durationMs: number;
  /** Mix volume (0–1). Keep low so narration sits on top. */
  volume: number;
}

export const MUSIC: MusicTrack = {
  prompt:
    "Minimal ambient underscore for a premium software brand film. Soft evolving synth pad, a few sparse warm piano or bell notes, gentle low sub pulse, lots of space and air. Calm, restrained, sophisticated. No drum buildup, no arpeggio hook, no cheesy melody, no advertisement or corporate-stock vibe, no big drop. Purely instrumental, no vocals, no speech. Thirty seconds, quiet and even throughout.",
  durationMs: 30000,
  volume: 0.11,
};

export const MUSIC_AUDIO_SUBDIR = "audio/linkedin-sequencer-music";
export const MUSIC_FILENAME = "background.mp3";

export function musicPath(): string {
  return `${MUSIC_AUDIO_SUBDIR}/${MUSIC_FILENAME}`;
}
